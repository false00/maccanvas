cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(aac_to_wav C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
file(TO_CMAKE_PATH "${FDK_ROOT}" FDK_ROOT)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory("${FDK_ROOT}" fdk EXCLUDE_FROM_ALL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../DISTRIBUTION-NOTICES.txt" AAC_NOTICES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" GLUE_LICENSE)
file(READ "${FDK_ROOT}/NOTICE" FDK_LICENSE)
string(APPEND AAC_NOTICES "\n${GLUE_LICENSE}\n${FDK_LICENSE}")
configure_file(licenses.h.in licenses.h @ONLY)
add_executable(aac-to-wav main.cpp)
target_include_directories(aac-to-wav PRIVATE "${FDK_ROOT}/libAACdec/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(aac-to-wav PRIVATE /utf-8)
target_link_libraries(aac-to-wav PRIVATE fdk-aac)
install(TARGETS aac-to-wav RUNTIME DESTINATION bin)
